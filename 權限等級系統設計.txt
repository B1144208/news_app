-- ===== 查看當前用戶權限等級 =====
SELECT 
    u.user_id,
    u.user_account,
    u.user_name,
    u.user_level,
    p.level_name,
    p.manage_data,
    p.add_manager,
    p.inherited_position
FROM user_profile u
LEFT JOIN permission_level p ON u.user_level = p.level_id
ORDER BY u.user_level DESC;
